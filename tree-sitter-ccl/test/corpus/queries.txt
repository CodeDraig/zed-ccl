CCL select with plan join
==================

SELECT INTO "nl:"
  PERSON_ID = P.PERSON_ID,
  COUNT(SEQ)
FROM PERSON P, ENCOUNTER E
PLAN P WHERE P.ACTIVE_IND = 1
JOIN E WHERE E.PERSON_ID = P.PERSON_ID
GROUP BY P.PERSON_ID
HAVING COUNT(SEQ) > 0
ORDER BY P.PERSON_ID DESC
WITH MAXREC = 100, NOCOUNTER
GO

==================
Chained nested set operations with outer clauses
==================

SELECT PERSON_ID = P.PERSON_ID
FROM PERSON P
WHERE P.ACTIVE_IND = 1
UNION ALL (
  SELECT PERSON_ID = E.PERSON_ID
  FROM ENCOUNTER E
  WHERE E.ACTIVE_IND = 1
  INTERSECT (
    SELECT PERSON_ID = O.PERSON_ID
    FROM ORDERS O
    WHERE O.ACTIVE_IND = 1
  )
)
ORDER BY PERSON_ID
DETAIL
  COL 0 PERSON_ID
WITH RDBUNION
GO

Subquery and documented ANSI join block

SELECT P.PERSON_ID
FROM PERSON P,
  (LEFT JOIN ENCOUNTER E ON (E.PERSON_ID = P.PERSON_ID))
PLAN P WHERE EXISTS (
  SELECT O.ORDER_ID
  FROM ORDERS O
  WHERE O.PERSON_ID = P.PERSON_ID
)
JOIN E
GO
