WHEN MATCHED THEN (
  UPDATE SET T.VALUE = S.VALUE WHERE T.ID = S.ID
)
WHEN NOT MATCHED THEN (
  INSERT INTO TARGET (ID, VALUE) VALUES (S.ID, S.VALUE)
)
WITH NOCOUNTER
